  1256→  1256→// Simple markdown renderer
  1257→  1257→function MarkdownRenderer({ content }: { content: string }) {
  1258→  1258→  const renderMarkdown = (text: string) => {
  1259→  1259→    const lines = text.split("\n");
  1260→  1260→    const elements: React.ReactNode[] = [];
  1261→  1261→    let inCodeBlock = false;
  1262→  1262→    let codeContent: string[] = [];
  1263→  1263→

  1265→  1265→      if (line.startsWith("```")) {
  1266→  1266→        if (inCodeBlock) {

  1268→  1268→            <pre
  1269→  1269→              key={`code-${idx}`}

  1273→  1273→            </pre>

  1277→  1277→        } else {
  1278→  1278→          inCodeBlock = true;
  1279→  1279→        }
  1280→  1280→        return;
  1281→  1281→      }
  1282→  1282→

  1290→  1290→          <h3

  1299→  1299→          <h2
  1300→  1300→            key={idx}
  1301→  1301→            className="text-xl font-bold mt-6 mb-3 text-slate-900 dark:text-white border-b pb-1"

  1335→  1335→          <p

  1347→  1347→
  1348→  1348→  const renderInlineFormat = (text: string): React.ReactNode => {
  1349→  1349→    const parts: React.ReactNode[] = [];
  1350→  1350→    let remaining = text;
  1351→  1351→    let key = 0;
  1352→  1352→
  1353→  1353→    while (remaining.length > 0) {
  1354→  1354→      const boldMatch = remaining.match(/\*\*(.+?)\*\*/);
  1355→  1355→      const codeMatch = remaining.match(/`([^`]+)`/);
  1356→  1356→      const linkMatch = remaining.match(/\[([^\]]+)\]\(([^)]+)\)/);
  1357→  1357→
  1358→  1358→      const matches = [
  1359→  1359→        boldMatch
  1360→  1360→          ? { type: "bold" as const, match: boldMatch, index: boldMatch.index! }
  1361→  1361→          : null,
  1362→  1362→        codeMatch
  1363→  1363→          ? { type: "code" as const, match: codeMatch, index: codeMatch.index! }
  1364→  1364→          : null,
  1365→  1365→        linkMatch
  1366→  1366→          ? { type: "link" as const, match: linkMatch, index: linkMatch.index! }
  1367→  1367→          : null,
  1368→  1368→      ].filter(Boolean) as Array<{
  1369→  1369→        type: "bold" | "code" | "link";
  1370→  1370→        match: RegExpMatchArray;
  1371→  1371→        index: number;
  1372→  1372→      }>;
