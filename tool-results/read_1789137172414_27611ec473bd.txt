  1256→// Simple markdown renderer
  1257→function MarkdownRenderer({ content }: { content: string }) {
  1258→  const renderMarkdown = (text: string) => {
  1259→    const lines = text.split("\n");
  1260→    const elements: React.ReactNode[] = [];
  1261→    let inCodeBlock = false;
  1262→    let codeContent: string[] = [];
  1263→
  1264→    lines.forEach((line, idx) => {
  1265→      if (line.startsWith("```")) {
  1266→        if (inCodeBlock) {
  1267→          elements.push(
  1268→            <pre
  1269→              key={`code-${idx}`}
  1270→              className="bg-slate-100 dark:bg-slate-900 p-4 rounded-lg overflow-x-auto my-4"
  1271→            >
  1272→              <code className="text-sm">{codeContent.join("\n")}</code>
  1273→            </pre>
  1274→          );
  1275→          codeContent = [];
  1276→          inCodeBlock = false;
  1277→        } else {
  1278→          inCodeBlock = true;
  1279→        }
  1280→        return;
  1281→      }
  1282→
  1283→      if (inCodeBlock) {
  1284→        codeContent.push(line);
  1285→        return;
  1286→      }
  1287→
  1288→      if (line.startsWith("### ")) {
  1289→        elements.push(
  1290→          <h3
  1291→            key={idx}
  1292→            className="text-lg font-bold mt-6 mb-2 text-slate-900 dark:text-white"
  1293→          >
  1294→            {line.slice(4)}
  1295→          </h3>
  1296→        );
  1297→      } else if (line.startsWith("## ")) {
  1298→        elements.push(
  1299→          <h2
  1300→            key={idx}
  1301→            className="text-xl font-bold mt-6 mb-3 text-slate-900 dark:text-white border-b pb-1"
  1302→          >
  1303→            {line.slice(3)}
  1304→          </h2>
  1305→        );
  1306→      } else if (line.startsWith("# ")) {
  1307→        elements.push(
  1308→          <h1
  1309→            key={idx}
  1310→            className="text-2xl font-bold mt-6 mb-3 text-slate-900 dark:text-white"
  1311→          >
  1312→            {line.slice(2)}
  1313→          </h1>
  1314→        );
  1315→      } else if (line.startsWith("- ")) {
  1316→        elements.push(
  1317→          <li key={idx} className="ml-6 list-disc text-slate-700 dark:text-slate-300">
  1318→            {renderInlineFormat(line.slice(2))}
  1319→          </li>
  1320→        );
  1321→      } else if (/^\d+\.\s/.test(line)) {
  1322→        elements.push(
  1323→          <li key={idx} className="ml-6 list-decimal text-slate-700 dark:text-slate-300">
  1324→            {renderInlineFormat(line.replace(/^\d+\.\s/, ""))}
  1325→          </li>
  1326→        );
  1327→      } else if (line.startsWith("---")) {
  1328→        elements.push(
  1329→          <hr key={idx} className="my-4 border-slate-200 dark:border-slate-700" />
  1330→        );
  1331→      } else if (line.trim() === "") {
  1332→        elements.push(<div key={idx} className="h-2" />);
  1333→      } else {
  1334→        elements.push(
  1335→          <p
  1336→            key={idx}
  1337→            className="text-slate-700 dark:text-slate-300 leading-relaxed my-1"
  1338→          >
  1339→            {renderInlineFormat(line)}
  1340→          </p>
  1341→        );
  1342→      }
  1343→    });
  1344→
  1345→    return elements;
  1346→  };
  1347→
  1348→  const renderInlineFormat = (text: string): React.ReactNode => {
  1349→    const parts: React.ReactNode[] = [];
  1350→    let remaining = text;
  1351→    let key = 0;
  1352→
  1353→    while (remaining.length > 0) {
  1354→      const boldMatch = remaining.match(/\*\*(.+?)\*\*/);
  1355→      const codeMatch = remaining.match(/`([^`]+)`/);
  1356→      const linkMatch = remaining.match(/\[([^\]]+)\]\(([^)]+)\)/);
  1357→
  1358→      const matches = [
  1359→        boldMatch
  1360→          ? { type: "bold" as const, match: boldMatch, index: boldMatch.index! }
  1361→          : null,
  1362→        codeMatch
  1363→          ? { type: "code" as const, match: codeMatch, index: codeMatch.index! }
  1364→          : null,
  1365→        linkMatch
  1366→          ? { type: "link" as const, match: linkMatch, index: linkMatch.index! }
  1367→          : null,
  1368→      ].filter(Boolean) as Array<{
  1369→        type: "bold" | "code" | "link";
  1370→        match: RegExpMatchArray;
  1371→        index: number;
  1372→      }>;
  1373→
  1374→      if (matches.length === 0) {
  1375→        parts.push(<span key={key++}>{remaining}</span>);
  1376→        break;
  1377→      }
  1378→
  1379→      matches.sort((a, b) => a.index - b.index);
  1380→      const first = matches[0];
  1381→
  1382→      if (first.index > 0) {
  1383→        parts.push(<span key={key++}>{remaining.slice(0, first.index)}</span>);
  1384→      }
  1385→
  1386→      if (first.type === "bold") {
  1387→        parts.push(
  1388→          <strong key={key++} className="font-bold text-slate-900 dark:text-white">
  1389→            {first.match[1]}
  1390→          </strong>
  1391→        );
  1392→      } else if (first.type === "code") {
  1393→        parts.push(
  1394→          <code
  1395→            key={key++}
  1396→            className="bg-slate-100 dark:bg-slate-800 px-1 py-0.5 rounded text-sm text-emerald-700 dark:text-emerald-400"
  1397→          >
  1398→            {first.match[1]}
  1399→          </code>
  1400→        );
  1401→      } else if (first.type === "link") {
  1402→        parts.push(
  1403→          <a
  1404→            key={key++}
  1405→            href={first.match[2]}
  1406→            target="_blank"
  1407→            rel="noopener noreferrer"
  1408→            className="text-emerald-600 hover:underline inline-flex items-center gap-0.5"
  1409→          >
  1410→            {first.match[1]}
  1411→            <ExternalLink className="h-3 w-3" />
  1412→          </a>
  1413→        );
  1414→      }
  1415→
  1416→      remaining = remaining.slice(first.index + first.match[0].length);
  1417→    }
  1418→
  1419→    return parts;
  1420→  };
  1421→
  1422→  return <div className="space-y-1">{renderMarkdown(content)}</div>;
  1423→}
